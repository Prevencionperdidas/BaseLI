function refrescarCamposPendientes(){

  'transporteRecepcion',
  'patenteRecepcion',
  'guia'

];

  campos.forEach(id => {

    const campo =
      document.getElementById(id);

    if(!campo) return;

    const vacio =

      campo.value.trim() === '' ||

      campo.value === 'Seleccione';

    campo.classList.remove(
      'campoPendiente',
      'campoCompleto'
    );

      vacio
        ? 'campoPendiente'

    );

  });

}

function cargarDetalleRecepcion(texto) {

    texto.split('\n');

  let patente = '';

    const c = linea.split('\t');

    if (
      c[0] === 'HORA' &&
      c[2]
    ) {

      transporte =
        c[2].trim();

    }

    if (

      c[0] === 'PPU' ||

      c[1] === 'PPU' ||

      c[0] === 'PATENTE' ||

      c[1] === 'PATENTE'

    ) {

      patente =
        (c[2] || c[1] || '')
          .trim();

    }

    if (
      c[0] === 'N°' &&
      c[1] === 'OLPN'
    ) {

      inicioDetalle =
        indice + 1;

    }

  });

  if (inicioDetalle === -1) {
    return;
  }

  document.getElementById(
    'transporteRecepcion'
  ).value = transporte;

  document.getElementById(
    'patenteRecepcion'
  ).value = patente;

  for (
    let i = inicioDetalle;
    i < lineas.length;
    i++
  ) {

    const c =
      lineas[i].split('\t');

    if (!c[1]) {
      continue;
    }

    const iso =
      String(c[2] || '')
        .trim();

    const comuna =
      String(c[6] || '')
        .trim();

    if (!grupos[iso]) {

      grupos[iso] = {

        comuna: comuna,

        detalle: []

      };

      olpn: c[1] || '',

function filtrarDetalle(){

  const texto =
    document
      .getElementById(
        'txtBuscarDetalle'
      )
      .value
      .toUpperCase();

  const filas =
    document.querySelectorAll(
      '[data-olpn]'
    );

  filas.forEach(fila => {

    const olpn =
      (fila.dataset.olpn || '')
        .toUpperCase();

    const sku =
      (fila.dataset.sku || '')
        .toUpperCase();

    const visible =

      olpn.includes(texto) ||

      sku.includes(texto);

    fila.style.display =
      visible
        ? ''
        : 'none';

  });

}

function cargarGeosort(){

  const archivo =
    document.getElementById(
      'archivoGeosort'
    ).files[0];

  if(!archivo){

    alert(
      'Seleccione un archivo'
    );

    return;

  }

  mostrarCarga();

  const reader =
    new FileReader();

  reader.onload = function(e){

    const base64 =
      e.target.result
        .split(',')[1];

    google.script.run

      .withSuccessHandler(
        function(){

          ocultarCarga();

          cerrarModalGeosort();

          alert(
            '✅ Archivo cargado correctamente'
          );

        }
      )

      .withFailureHandler(
        function(error){

          ocultarCarga();

          alert(
            error.message || error
          );

        }
      )

      .subirArchivoDrive(
        archivo.name,
        base64,
        archivo.type
      );

  };

  reader.readAsDataURL(
    archivo
  );

}

function cargarGestion() {

        document.getElementById(
          'contenedorGestion'
        );

      if (!contenedor) {
        return;
      }

      contenedor.innerHTML = '';

      if (
        !Array.isArray(datos) ||
        datos.length === 0

        contenedor.innerHTML =
          '<div style="padding:20px;color:#64748b;font-weight:bold;">' +
          'No existen ISO para mostrar' +
          '</div>';

        return;
      }

      datos.forEach(function(item) {

        let html = '';

        const isoSeguro =
          String(item.iso || '')
            .replace(/'/g, "\\'")

        html +=
          '<button ' +
          'class="cardGestion elementoBusqueda ' +

          (item.checkinBO
            ? 'enviada '
            : ''

            ? 'incidencia '
            : ''
          ) +

          '" ' +
          'data-iso="' + isoSeguro + '" ' +
          'data-sku="' + (item.sku || []).join(' ') + '" ' +

          '</div>';

          '<div class="comunaGestionCard">' +
          (item.comuna || 'SIN COMUNA') +
          '</div>';

        html +=
          '<div class="bultosGestionCard">' +
          (
            item.tienePlanificacion
              ? '📦 ' +
                item.olpnRecepcionados +
                '/' +
                item.olpnPlanificados
              : '📦 S/D'
          ) +
          '</div>';

        if(item.checkinBO){

          html +=
            '<div class="badgeEnviado">' +
            '✅ ENVIADO' +
            '</div>';

        }else if(item.tieneIncidencia){

          html +=
            '<div class="badgePrioridad">' +

            '</div>';

              '<div class="badgeValor">' +
                item.casoLi +

llenarCombo(

function buscarGestion(){

  const texto =
    document
      .getElementById(
        'txtBusquedaGestion'
      )
      .value
      .trim()
      .toUpperCase();

  const terminos =
    texto
      .replace(/,/g,' ')
      .split(/\s+/)
      .filter(Boolean);

  const elementos =
    document.querySelectorAll(
      '.elementoBusqueda'
    );

  elementos.forEach(item => {

    const contenido =
      item.innerText
        .toUpperCase();

    if(terminos.length === 0){

      item.style.display = '';
      return;

    }

    const encontrado =
      terminos.some(termino =>
        contenido.includes(
          termino
        )
      );

    item.style.display =
      encontrado
        ? ''
        : 'none';

  });

}

  if(
    tag === 'INPUT' ||
    tag === 'TEXTAREA'
  ){

    if(
      e.target.type !== 'date' &&
      e.target.type !== 'time' &&
      e.target.type !== 'number'
    ){

      e.target.value =
        e.target.value.toUpperCase();

    }

  }

});

function obtenerColorRepite(repite){

  repite = Number(repite || 0);

  if(repite <= 1){
    return '#e8f1ff';
  }

  if(repite === 2){
    return '#cfe2ff';
  }

  if(repite === 3){
    return '#9ec5fe';
  }

  if(repite === 4){
    return '#6ea8fe';
  }

  if(repite === 5){
    return '#3d8bfd';
  }

  if(repite === 6){
    return '#0d6efd';
  }

  if(repite === 7){
    return '#0a58ca';
  }

  return '#084298';
}

            <th>REPITE</th>
            <th>COMUNA</th>
            <th>BULTOS</th>

          </tr>

        </thead>

        <tbody>

      `;

      datos.forEach(r => {

        html += `

          <tr>

            <td>${r.iso}</td>

            <td>${r.repite}</td>

            <td>${r.comuna}</td>

            <td>${r.bultos}</td>

          </tr>

        `;

      });

          </tbody>
        </table>
      `;

      document.getElementById(
        'detalleManifiestoGrid'

      document.getElementById(
        'modalDetalleManifiesto'
      ).style.display = 'flex';

      alert(
        error.message || error

      idManifiesto
    );
